**Step 1: Create PluginManager struct**

Create src/plugin/manager.zig:
```zig
// Plugin manager for plugin discovery and lifecycle management
//
// Manages compile-time plugin registry, handles activation/deactivation,
// routes events to active plugin

const std = @import("std");
const Plugin = @import("interface.zig").Plugin;
const PluginContext = @import("interface.zig").PluginContext;
const PluginEvent = @import("interface.zig").PluginEvent;
const HalValue = @import("interface.zig").HalValue;

/// Plugin manager
pub const PluginManager = struct {
    allocator: std.mem.Allocator,
    /// All available plugins (compile-time registry)
    registry: []const *const Plugin,
    /// Currently active plugin (null = none active)
    active: ?*const Plugin,
    /// Plugin context for active plugin
    context: PluginContext,
    /// User context for plugin (plugin-specific data)
    user_context: ?*anyopaque,

    /// Initialize plugin manager with plugin registry
    pub fn init(
        allocator: std.mem.Allocator,
        registry: []const *const Plugin,
        getValueFn: *const fn ([]const u8) ?HalValue,
        subscribeFn: *const fn ([]const u8, *const fn ([]const u8, HalValue) void) anyerror!void,
        logInfoFn: *const fn ([]const u8) void,
        logErrorFn: *const fn ([]const u8) void,
    ) PluginManager {
        return .{
            .allocator = allocator,
            .registry = registry,
            .active = null,
            .context = .{
                .allocator = allocator,
                .user_context = null,
                .getValue = getValueFn,
                .subscribe = subscribeFn,
                .logInfo = logInfoFn,
                .logError = logErrorFn,
            },
            .user_context = null,
        };
    }

    /// Get list of all available plugins
    pub fn listPlugins(self: *const PluginManager) []const *const Plugin {
        return self.registry;
    }

    /// Get active plugin (null if none)
    pub fn getActive(self: *const PluginManager) ?*const Plugin {
        return self.active;
    }

    /// Activate a plugin by name
    pub fn activate(self: *PluginManager, name: []const u8) !void {
        // Deactivate current plugin if any
        if (self.active) |plugin| {
            self.deactivate();
        }

        // Find plugin by name
        for (self.registry) |plugin| {
            if (std.mem.eql(u8, plugin.name, name)) {
                self.active = plugin;
                self.context.user_context = null;

                // Initialize plugin
                plugin.init(&self.context) catch |err| {
                    self.active = null;
                    return err;
                };

                return;
            }
        }

        return error.PluginNotFound;
    }

    /// Deactivate active plugin
    pub fn deactivate(self: *PluginManager) void {
        if (self.active) |plugin| {
            plugin.deinit(&self.context);
            self.active = null;
            self.context.user_context = null;
        }
    }

    /// Route event to active plugin
    pub fn handleEvent(self: *PluginManager, event: PluginEvent) bool {
        if (self.active) |plugin| {
            return plugin.handleEvent(&self.context, event);
        }
        return false;
    }

    /// Request render from active plugin
    pub fn renderActive(self: *PluginManager, ctx: anytype) !void {
        if (self.active) |plugin| {
            plugin.render(&self.context, ctx);
        }
    }

    /// Check if any plugin is active
    pub fn hasActive(self: *const PluginManager) bool {
        return self.active != null;
    }

    /// Get active plugin name (null if none)
    pub fn getActiveName(self: *const PluginManager) ?[]const u8 {
        if (self.active) |plugin| {
            return plugin.name;
        }
        return null;
    }

    /// Set user context for plugin
    pub fn setUserContext(self: *PluginManager, ctx: ?*anyopaque) void {
        self.user_context = ctx;
        self.context.user_context = ctx;
    }
};

/// Error types
pub const PluginManagerError = error{
    PluginNotFound,
    InitFailed,
};
```
