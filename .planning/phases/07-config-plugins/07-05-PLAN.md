## Task 1: Create Hello Plugin

**Files:**
- Create: `src/plugin/builtin/hello_plugin.zig`

**Step 1: Create hello plugin implementation**

Create src/plugin/builtin/hello_plugin.zig:
```zig
// Hello Plugin - Minimal example plugin for haltune
//
// Demonstrates basic plugin API usage: init, deinit, render, handleEvent

const std = @import("std");
const vxfw = @import("vaxis").vxfw;
const interface = @import("../interface.zig");

/// Plugin state
const HelloState = struct {
    counter: u32 = 0,
    last_message: []const u8 = "Hello from plugin!",
};

var plugin_state: HelloState = .{};

fn init(ctx: *interface.PluginContext) anyerror!void {
    ctx.logInfo("Hello plugin initialized!");
    plugin_state.counter = 0;
}

fn deinit(ctx: *interface.PluginContext) void {
    _ = ctx;
    plugin_state.counter = 0;
}

fn render(ctx: *interface.PluginContext, vctx: vxfw.Context) anyerror!void {
    _ = ctx;

    const width = vctx.constraint().max.width;
    const height = vctx.constraint().max.height;

    // Draw centered message
    const msg = try std.fmt.allocPrint(ctx.allocator,
        \\[ Hello Plugin ({d} presses) ]
        ++ plugin_state.counter);

    const msg_len = msg.len;
    const x = if (msg_len < width) @divTrunc(width - msg_len, 2) else 0;
    const y = @divTrunc(height, 2);

    _ = try vctx.write(.{
        .text = msg,
        .offset = .{ .x = @intFromFloat(@as(f32, x)), .y = @as(u16, y) },
        .style = .{ .bold = true },
    });

    // Draw instructions
    const instructions = \\Press 'q' to deactivate plugin\\;
    const inst_y = @as(u16, y + 2);
    _ = try vctx.write(.{
        .text = instructions,
        .offset = .{ .x = 0, .y = inst_y },
        .style = .{ .dim = true },
    });
}

fn handleEvent(ctx: *interface.PluginContext, event: interface.PluginEvent) bool {
    _ = ctx;

    switch (event) {
        .key_press => |kp| {
            if (kp.codepoint == 'q') {
                // Don't handle 'q' here - let it deactivate plugin
                return false;
            }

            // Count key presses
            plugin_state.counter += 1;
            return true; // We handled it
        },
        .focus => {
            // Plugin gained focus
            plugin_state.counter = 0;
        },
        else => {},
    }

    return false; // Don't handle other events
}

pub const hello_plugin: interface.Plugin = .{
    .name = "Hello",
    .version = "1.0.0",
    .description = "A minimal example plugin",
    .init = init,
    .deinit = deinit,
    .render = render,
    .handleEvent = handleEvent,
};
```
