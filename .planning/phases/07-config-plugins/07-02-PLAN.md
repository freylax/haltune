/// Get text style for origin type
fn originStyle(origin: Origin) vxfw.Text.Style {
    return switch (origin) {
        .hal_file => .{ .fg = .{ .index = 4 } }, // Blue
        .ini_file => .{ .fg = .{ .index = 2 } }, // Green
        .runtime_modified => .{ .fg = .{ .index = 3 } }, // Yellow
        .default_value, .none => .{}, // Default (no color)
    };
}

**Step 3: Add formatOrigin() helper function**

```zig
/// Format origin as display string
fn formatOrigin(allocator: std.mem.Allocator, origin: ItemOrigin) ![]const u8 {
    if (origin.origin == .none or origin.origin == .default_value) {
        return allocator.dupe(u8, "");
    }

    // Show file basename for .hal/.ini origins
    if (origin.file_path) |file_path| {
        const basename = std.fs.path.basename(file_path);
        return std.fmt.allocPrint(allocator, "{s}", .{basename});
    }

    return allocator.dupe(u8, "runtime");
}
```

Update the cell rendering loop:
```zig
fn drawColumnCells(self: *DataTable, ctx: vxfw.Context) !void {
    const width = ctx.constraint().max.width;
    const name_width = @as(f32, width) * 0.30;
    const type_width = @as(f32, width) * 0.12;
    const dir_width = @as(f32, width) * 0.10;
    const origin_width = @as(f32, width) * 0.18;
    const value_width = @as(f32, width) * 0.30;

    for (self.items.items, 0..) |item, row_idx| {
        const y = @as(u16, row_idx + 1); // +1 for header
        var x: u16 = 0;

        // Name column
        const name_style = if (row_idx == self.cursor_idx) .{ .reverse = true } else .{};
        _ = try ctx.write(.{ .text = item.name }, .{
            .width = @intFromFloat(name_width),
            .offset = x,
            .style = name_style,
        });
        x += @intFromFloat(name_width);

        // Type column
        const type_str = item.hal_type.toString();
        _ = try ctx.write(.{ .text = type_str }, .{
            .width = @intFromFloat(type_width),
            .offset = x,
        });
        x += @intFromFloat(type_width);

        // Direction column
        const dir_str = if (item.direction) |dir|
            dir.toString()
        else
            "";
        _ = try ctx.write(.{ .text = dir_str }, .{
            .width = @intFromFloat(dir_width),
            .offset = x,
        });
        x += @intFromFloat(dir_width);

        // Origin column (NEW)
        const origin_str = formatOrigin(ctx.arena, item.origin) catch "";
        const origin_style = originStyle(item.origin.origin);
        _ = try ctx.write(.{ .text = origin_str }, .{
            .width = @intFromFloat(origin_width),
            .offset = x,
            .style = origin_style,
        });
        x += @intFromFloat(origin_width);

        // Value column
        const value_str = formatHalValue(item.value);
        _ = try ctx.write(.{ .text = value_str }, .{
            .width = @intFromFloat(value_width),
            .offset = x,
        });
    }
}
```
