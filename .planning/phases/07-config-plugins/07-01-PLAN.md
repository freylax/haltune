## Task 3: Parse Configuration Files at Startup

**Files:**
- Modify: `src/tui/model.zig`

**Step 1: Add parser imports at top of model.zig**

```zig
const hal_parser = @import("../config/hal_parser.zig");
const ini_parser = @import("../config/ini_parser.zig");
```

**Step 2: Add parseConfigFiles() method to Model**

After Model.init(), add method to parse config files:
```zig
/// Parse configuration files and populate origin tracker
fn parseConfigFiles(self: *Model) !void {
    const allocator = self.allocator;

    // Parse .hal files
    for (self.config.hal_files.items) |file_path| {
        std.log.info("Parsing .hal file: {s}", .{file_path});
        const commands = try hal_parser.parseHalFile(allocator, file_path);

        // Process commands and populate origin tracker
        for (commands.items) |cmd| {
            switch (cmd) {
                .setp => |setp_cmd| {
                    try self.store.origin_tracker.setParamOrigin(
                        allocator,
                        setp_cmd.name,
                        .{ .fromHalFile = .{ .allocator = allocator, .file_path = file_path, .line = setp_cmd.line },
                    );
                },
                .net => |net_cmd| {
                    // Track signal origin from net command
                    try self.store.origin_tracker.setSignalOrigin(
                        allocator,
                        net_cmd.signal_name,
                        .{ .fromHalFile = .{ .allocator = allocator, .file_path = file_path, .line = net_cmd.line },
                    );
                },
                else => {}, // Other commands don't need origin tracking yet
            }
        }

        // Free parsed commands
        hal_parser.freeCommands(allocator, commands);
    }

    // Parse .ini files
    for (self.config.ini_files.items) |file_path| {
        std.log.info("Parsing .ini file: {s}", .{file_path});
        const entries = try ini_parser.parseIniFile(allocator, file_path);

        // Process entries and populate origin tracker
        for (entries.items) |entry| {
            switch (entry) {
                .key_value => |kv| {
                    // Check if this is a HAL parameter reference
                    if (std.mem.startsWith(u8, kv.key, "HAL_") or
                        std.mem.startsWith(u8, kv.key, "PARAM_")) {
                        // Map [SECTION]VARIABLE to param name
                        const param_name = try std.fmt.allocPrint(allocator, "{s}.{s}", .{ kv.section, kv.key });
                        try self.store.origin_tracker.setParamOrigin(
                            allocator,
                            param_name,
                            .{ .fromIniFile = .{ .allocator = allocator, .file_path = file_path, .section = kv.section, .variable = kv.key, .line = kv.line },
                        );
                    }
                },
                else => {},
            }
        }

        // Free parsed entries
        ini_parser.freeEntries(allocator, entries);
    }
}
```
