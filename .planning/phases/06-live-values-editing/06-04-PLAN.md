```zig
/// Get full precision value string for a HAL item
///
/// Returns a formatted string showing the item's name, type, and full
/// precision value. Unlike the compact format used in tree view (6 chars),
/// this uses full precision for floats and word-format for bits.
///
/// Parameters:
///   - allocator: Arena allocator from DrawContext
///   - node: TreeNode representing the HAL item
///
/// Returns:
///   - Formatted string "name: TYPE value" or error if fetch fails
///
/// Example output:
///   - "motion.digital-in-00: BIT TRUE"
///   - "motion.analog-in-00: FLOAT 3.14159265358979"
pub fn getFullValueString(self: *const Model, allocator: std.mem.Allocator, node: *const TreeNode) ![]const u8 {
    if (node.item_type == .component) {
        // Components don't have values
        return allocator.dupe(u8, node.full_name);
    }

    // Fetch value from store
    const value = self.store.getPin(node.full_name) catch
                  self.store.getSignal(node.full_name) catch
                  self.store.getParam(node.full_name) catch null;

    if (value) |v| {
        const type_str = switch (v) {
            .bit => "BIT",
            .float => "FLOAT",
            .s32 => "S32",
            .u32 => "U32",
        };

        const value_str = switch (v) {
            .bit => |b| if (b) "TRUE" else "FALSE",
            .float => |f| try std.fmt.allocPrint(allocator, "{d}", .{f}), // Full precision
            .s32 => |s| try std.fmt.allocPrint(allocator, "{d}", .{s}),
            .u32 => |u| try std.fmt.allocPrint(allocator, "{d}", .{u}),
        };

        return std.fmt.allocPrint(allocator, "{s}: {s} {s}", .{ node.full_name, type_str, value_str });
    } else {
        // No value available
        return std.fmt.allocPrint(allocator, "{s}: (no value)", .{node.full_name});
    }
}
```

WHY TRUE/FALSE for BIT: Words are clearer than ●/○ in status line text
