```zig
/// Format a HAL value for display in the value column
/// Uses compact formatting: ●/○ for BIT, 6-char precision for FLOAT/U32
/// Matches tree_view.zig formatting for consistency
fn formatHalValue(value: HalValue, allocator: std.mem.Allocator) ![]const u8 {
    return switch (value) {
        .bit => |v| if (v) "\xe2\x97\x8f" else "\xe2\x97\x8b", // UTF-8 for ●/○
        .float => |v| std.fmt.allocPrint(allocator, "{d:.6}", .{v}) catch "ERR",
        .s32 => |v| std.fmt.allocPrint(allocator, "{d}", .{v}) catch "ERR",
        .u32 => |v| std.fmt.allocPrint(allocator, "{d}", .{v}) catch "ERR",
    };
}
```

```zig
// ... existing name, type, direction rendering ...

// Value column
{
    const value_col_width = @as(u16, @intCast(floatFromInt(surface.size.width) * 0.30));
    const col_x = name_col_end + type_col_width + dir_col_width + 2; // +2 for padding

    if (col_x + value_col_width <= surface.size.width) {
        // Column header
        if (row == 0) {
            const header = "Value";
            for (header, 0..) |c, i| {
                if (col_x + i < surface.size.width) {
                    surface.writeCell(col_x + i, row, .{
                        .char = .{ .grapheme = &[_]u8{c}, .width = 1 },
                        .style = .{ .bold = true }
                    });
                }
            }
            row += 1;
        }

        // Draw value for each item
        for (self.items.items, 0..) |item, item_idx| {
            if (row >= surface.size.height) break;

            // Fetch value from store
            const value = blk: {
                if (item.item_type == .pin) {
                    break :blk self.store.getPin(item.name) catch null;
                } else if (item.item_type == .signal) {
                    break :blk self.store.getSignal(item.name) catch null;
                } else if (item.item_type == .param) {
                    break :blk self.store.getParam(item.name) catch null;
                }
                break :blk null;
            };

            // Format and render value
            if (value) |v| {
                const value_str = formatHalValue(v, ctx.arena) catch "ERR";
                const value_width = ctx.stringWidth(value_str);

                // Right-align value in column
                const value_x = @min(col_x + value_col_width -| value_width, col_x + value_col_width);
                var char_iter = ctx.graphemeIterator(value_str);
                var char_col: u16 = value_x;

                while (char_iter.next()) |char| {
                    const grapheme = char.bytes(value_str);
                    const grapheme_width: u8 = @intCast(ctx.stringWidth(grapheme));
                    if (char_col + grapheme_width <= surface.size.width) {
                        surface.writeCell(char_col, row, .{
                            .char = .{ .grapheme = grapheme, .width = grapheme_width },
                            .style = .{},
                        });
                        char_col += grapheme_width;
                    }
                }
            } else {
                // No value available
                surface.writeCell(col_x, row, .{
                    .char = .{ .grapheme = "-", .width = 1 },
                    .style = .{ .dim = true },
                });
            }

            row += 1;
        }
    }
}
```
